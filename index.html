async function permanentDelete(id, vehicleNumber, photoPath){ const ok = confirm( "Permanently delete " + vehicleNumber + "?\n\nVehicle record and its photo will be permanently deleted." ); if(!ok) return; try{ // 1. Delete photo from Storage if(photoPath){ const { error: photoError } = await supabaseClient .storage .from("vehicle-photos") .remove([photoPath]); if(photoError){ console.error("Photo delete error:", photoError); } } // 2. Delete vehicle record from database const { error: dbError } = await supabaseClient .from("vehicles") .delete() .eq("id", id); if(dbError){ throw dbError; } alert( "Vehicle and photo deleted permanently." ); // 3. Refresh dashboard await loadVehicles(); }catch(error){ console.error("Delete error:", error); alert( "Delete failed: " + (error.message || "Unknown error") ); } }
